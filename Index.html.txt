        // Render Action Queue
        function renderActionsList() {
            const container = document.getElementById('action-list-container');
            container.innerHTML = '';

            const filtered = STATE.actions.filter(act => {
                if(STATE.filterStatus === 'all') return true;
                return act.status === STATE.filterStatus;
            });

            document.getElementById('action-counter').innerText = `Showing ${filtered.length} of ${STATE.actions.length} items`;

            filtered.forEach(act => {
                let badgeColor = "bg-amber-500/10 text-amber-400 border-amber-500/20";
                if(act.status === 'Completed') badgeColor = "bg-emerald-500/10 text-emerald-400 border-emerald-500/20";
                if(act.status === 'In Progress') badgeColor = "bg-blue-500/10 text-blue-400 border-blue-500/20";

                const card = document.createElement('div');
                card.className = "bg-slate-900 border border-slate-800 hover:border-slate-700 rounded-2xl p-4 transition shadow-sm flex flex-col md:flex-row md:items-center justify-between gap-4";
                card.innerHTML = `
                    <div class="space-y-1.5 flex-1">
                        <div class="flex items-center gap-2 flex-wrap">
                            <span class="text-[10px] font-medium px-2.5 py-0.5 rounded-full border ${badgeColor}">${act.status}</span>
                            <span class="text-[10px] bg-slate-800 text-slate-300 px-2.5 py-0.5 rounded-full"><i class="fa-solid fa-user-tie mr-1 text-indigo-400"></i>${act.advisor}</span>

                            <span class="text-[10px] text-slate-400"><i class="fa-solid fa-calendar mr-1 text-slate-500"></i>Due: ${act.deadline}</span>
                        </div>
                        <h4 class="font-bold text-xs md:text-sm text-white">${act.title}</h4>
                        <div class="flex items-center gap-2 flex-wrap pt-1">
                            <span class="text-[10px] text-slate-400 font-medium">Milestones:</span>
                            ${act.milestones.map(m => `<span class="text-[10px] bg-slate-950 border border-slate-800 text-slate-300 px-2 py-0.5 rounded">${m}</span>`).join('')}
                        </div>
                    </div>
                    <div class="flex items-center gap-2 shrink-0">
                        <select onchange="updateActionStatus(${act.id}, this.value)" class="bg-slate-950 border border-slate-700 rounded-xl px-3 py-1.5 text-xs text-slate-200 focus:outline-none focus:border-indigo-500">
                            <option value="Not Started" ${act.status==='Not Started'?'selected':''}>Not Started</option>
                            <option value="In Progress" ${act.status==='In Progress'?'selected':''}>In Progress</option>
                            <option value="Completed" ${act.status==='Completed'?'selected':''}>Completed</option>
                        </select>
                        <button onclick="deleteAction(${act.id})" class="text-slate-500 hover:text-rose-400 p-2 rounded-xl hover:bg-slate-800 transition">
                            <i class="fa-solid fa-trash-can text-xs"></i>
                        </button>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function filterActions(status) {
            STATE.filterStatus = status;
            ['all', 'Not Started', 'In Progress', 'Completed'].forEach(s => {
                const btnId = s === 'all' ? 'filter-all' : `filter-${s.toLowerCase().replace(' ', '-')}`;
                const btn = document.getElementById(btnId);
                if(s === status) {
                    btn.className = "px-3 py-1 rounded-lg text-xs font-medium bg-indigo-600 text-white transition";
                } else {
                    btn.className = "px-3 py-1 rounded-lg text-xs font-medium bg-slate-800 text-slate-300 hover:bg-slate-700 transition";
                }
            });
            renderActionsList();
        }

        function updateActionStatus(id, newStatus) {
            const act = STATE.actions.find(a => a.id === id);
            if(act) {
                act.status = newStatus;
                renderActionsList();
                showToast("Action Updated", `Status changed to ${newStatus}`);
            }
        }

        function deleteAction(id) {
            STATE.actions = STATE.actions.filter(a => a.id !== id);
            renderActionsList();
            renderStrategySections();
            showToast("Action Removed", "Item deleted from queue");
        }

        function saveNewAction() {
            const title = document.getElementById('new-act-title').value.trim();
            const advisor = document.getElementById('new-act-advisor').value;
            const timeframe = document.getElementById('new-act-timeframe').value.trim() || 'Next 7 Days';
            const deadline = document.getElementById('new-act-deadline').value || new Date().toISOString().split('T')[0];
            const milestonesRaw = document.getElementById('new-act-milestones').value.trim();
            const milestones = milestonesRaw ? milestonesRaw.split(',').map(m => m.trim()) : ['Initial phase', 'Review & Execution'];

            if(!title) {
                showToast("Validation Error", "Please enter an action title.");
                return;
            }

            STATE.actions.unshift({
                id: Date.now(),
                title,
                advisor,
                timeframe,
                deadline,
                status: 'Not Started',
                milestones
            });

            closeNewActionModal();
            renderActionsList();
            renderStrategySections();
            showToast("Action Created", "New milestone added to queue.");

            // Clear inputs
            document.getElementById('new-act-title').value = '';
            document.getElementById('new-act-milestones').value = '';
        }

        function escapeHtml(text) {
            const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
            return text.replace(/[&<>"']/g, m => map[m]);
